/** * ERP 編碼系統 - 後端閃電極速版 (v3.5) */ var MAX_HISTORY = 30; function doPost(e) { try { var params = JSON.parse(e.postData.contents); var action = params.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === "getInitialData") { return response({ customers: getMaster(ss, "客戶主檔"), vendors: getMaster(ss, "廠商主檔"), projectHistory: getRecentHistory(ss, "客戶專案"), vendorHistory: getRecentHistory(ss, "廠商品號") }); } if (action === "getHistory") { var sheetName = (params.data && params.data.type === 'project') ? "客戶專案" : "廠商品號"; return response(getRecentHistory(ss, sheetName)); } if (action === "submitProject") return response(submitProject(ss, params.data)); if (action === "submitVendor") return response(submitVendor(ss, params.data)); if (action === "deleteRow") return response(deleteRow(ss, params.data)); if (action === "updateRow") return response(updateRow(ss, params.data)); return response("未知的動作指令: " + action, "error"); } catch (err) { return response(err.toString(), "error"); } } function getMaster(ss, name) { var sheet = ss.getSheetByName(name); if (!sheet) return []; var lastRow = sheet.getLastRow(); if (lastRow < 2) return []; var data = sheet.getRange(2, 1, lastRow - 1, 2).getDisplayValues(); var list = []; for (var i = 0; i < data.length; i++) { var id = data[i][0].trim(); var title = data[i][1].trim(); if (id !== "") list.push({ id: id, name: title }); } return list; } function getRecentHistory(ss, name) { var sheet = ss.getSheetByName(name); if (!sheet) return []; var lastRow = sheet.getLastRow(); if (lastRow < 2) return []; var startRow = Math.max(2, lastRow - MAX_HISTORY + 1); var data = sheet.getRange(startRow, 1, lastRow - startRow + 1, 6).getDisplayValues(); return data.filter(function(r) { return r[1] && r[1].trim() !== ""; }).reverse(); } // 閃電版：直接依據當前列數計算流水號，不跑迴圈，0.1秒完成寫入 function submitProject(ss, data) { var sheet = ss.getSheetByName("客戶專案"); var year = new Date().getFullYear().toString().substring(2); var cleanCustID = (data.customerID || "").toString().trim(); var amt = Number((data.amount || "0").toString().replace(/,/g, '')); var type = data.isCaseD === "是" ? "D" : (amt >= 8000000 ? "A" : (amt >= 1000000 ? "B" : "C")); var prefix = type + year + cleanCustID; var lastRow = sheet.getLastRow(); var count = (lastRow < 2) ? 1 : (lastRow - 1 + 1); // 簡易流水號基底 var neededLength = Math.max(1, 10 - prefix.length); var finalID = prefix + count.toString().padStart(neededLength, '0'); sheet.appendRow([new Date(), finalID, cleanCustID, amt, type, data.itemName || ""]); SpreadsheetApp.flush(); return finalID; } function submitVendor(ss, data) { var sheet = ss.getSheetByName("廠商品號"); var cat = ""; var prop = (data.property || "").toString(); var cleanVendorID = (data.vendorID || "").toString().trim(); if (prop.indexOf("連工帶料") !== -1) cat = "7A"; else if (prop.indexOf("工資") !== -1) cat = "7B"; else if (prop.indexOf("7C1") !== -1 || prop.indexOf("吊車") !== -1) cat = "7C1"; else if (prop.indexOf("7C2") !== -1 || prop.indexOf("運輸") !== -1) cat = "7C2"; var regCode = "00"; if (cat.indexOf("7C") === 0) { var north = ["台北", "桃園", "新竹", "宜蘭", "基隆", "離島", "花蓮", "台東"]; var central = ["台中", "苗栗", "彰化", "南投"]; var south = ["雲林", "嘉義", "台南", "高雄", "屏東"]; if (north.indexOf(data.city) !== -1) regCode = "02"; else if (central.indexOf(data.city) !== -1) regCode = "04"; else if (south.indexOf(data.city) !== -1) regCode = "07"; } var prefix = (cat === "7A" || cat === "7B") ? (cat + cleanVendorID) : (cat.substring(0, 3) + regCode); var lastRow = sheet.getLastRow(); var count = (lastRow < 2) ? 1 : (lastRow - 1 + 1); var finalID = cat.indexOf("7C") === 0 ? prefix + count.toString().padStart(3, '0') : prefix + count.toString().padStart(2, '0'); sheet.appendRow([new Date(), finalID, cleanVendorID, prop, data.city || "", data.itemName || ""]); SpreadsheetApp.flush(); return finalID; } function updateRow(ss, data) { var sheetName = data.type === 'project' ? "客戶專案" : "廠商品號"; var sheet = ss.getSheetByName(sheetName); var lastRow = sheet.getLastRow(); if (lastRow < 2) return "無資料可更新"; var targetCode = (data.code || "").toString().trim(); var values = sheet.getRange(2, 2, lastRow - 1, 1).getDisplayValues(); for (var i = values.length - 1; i >= 0; i--) { if (values[i][0].trim() === targetCode) { var rowNum = i + 2; sheet.getRange(rowNum, 4).setValue(data.newAmt); sheet.getRange(rowNum, 6).setValue(data.newName); return "更新成功"; } } return "找不到該筆資料"; } function deleteRow(ss, data) { var sheetName = data.type === 'project' ? "客戶專案" : "廠商品號"; var sheet = ss.getSheetByName(sheetName); var lastRow = sheet.getLastRow(); if (lastRow < 2) return "無資料可刪除"; var targetCode = (data.code || "").toString().trim(); var values = sheet.getRange(2, 2, lastRow - 1, 1).getDisplayValues(); for (var i = values.length - 1; i >= 0; i--) { if (values[i][0].trim() === targetCode) { sheet.deleteRow(i + 2); return "刪除成功"; } } return "找不到該筆資料"; } function response(d, s) { return ContentService.createTextOutput(JSON.stringify({status: s || "success", data: d})).setMimeType(ContentService.MimeType.JSON); }
